#!/usr/bin/env bash
# dav.sh USERVAR METHOD PATH : one DAV request as alice or bob with the app password (curl config on stdin).
# Prints "<METHOD> <path> auth=basic nutzer=<u> status=<code> fileid=<oc:fileid> permissions=<oc:permissions> exception=<s:exception>"
set -a; . C:/Users/Student/nextcloud-mcp-connector-worktrees/tasks-080/.env.nc35; set +a
u="$1"; m="$2"; p="$3"
pw="$NC_MCP_TEST_APP_PASSWORD"; [ "$u" = bob ] && pw="$NC_MCP_TEST_APP_PASSWORD2"
enc="$(printf '%s' "$p" | sed 's/ /%20/g')"
body=""
if [ "$m" = PROPFIND ]; then body='<?xml version="1.0"?><d:propfind xmlns:d="DAV:" xmlns:oc="http://owncloud.org/ns"><d:prop><oc:fileid/><oc:permissions/><d:resourcetype/></d:prop></d:propfind>'; fi
out="$(printf 'user = "%s:%s"\n' "$u" "$pw" | curl -s -m 60 -K - -X "$m" -H 'Depth: 0' ${body:+-H 'Content-Type: application/xml' --data "$body"} -w '\nHTTPSTATUS=%{http_code}' "$NC_MCP_URL/remote.php/dav/files/$u/$enc")"
code="$(printf '%s' "$out" | sed -n 's/^HTTPSTATUS=//p')"
fid="$(printf '%s' "$out" | grep -o '<oc:fileid>[0-9-]*' | head -n1 | sed 's/.*>//')"
perm="$(printf '%s' "$out" | grep -o '<oc:permissions>[A-Z]*' | head -n1 | sed 's/.*>//')"
exc="$(printf '%s' "$out" | grep -o '<s:exception>[^<]*' | head -n1 | sed 's/.*>//')"
echo "$m /$p auth=basic nutzer=$u status=$code${fid:+ fileid=$fid}${perm:+ permissions=$perm}${exc:+ exception=$exc}"
